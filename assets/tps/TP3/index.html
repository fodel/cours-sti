Partie A : Base de données

1. Création de la base de données et des tables (SQL)

-- Création de la base de données
CREATE DATABASE BDEvaluation;
USE BDEvaluation;

-- Création de la table 'testeur'
CREATE TABLE testeur (
    numPermis VARCHAR(8) PRIMARY KEY,
    nom VARCHAR(20) NOT NULL,
    prenom VARCHAR(20) NOT NULL,
    genre CHAR(1) NOT NULL
);

-- Création de la table 'modeleVoiture'
CREATE TABLE modeleVoiture (
    idModele INT PRIMARY KEY,
    libelle VARCHAR(20) NOT NULL
);

-- Création de la table 'evaluation'
CREATE TABLE evaluation (
    numPermis VARCHAR(8),
    idModele INT,
    dateTest DATETIME NOT NULL,
    securite INT NOT NULL,
    conduite INT NOT NULL,
    confort INT NOT NULL,
    PRIMARY KEY (numPermis, idModele),
    FOREIGN KEY (numPermis) REFERENCES testeur(numPermis) ON DELETE CASCADE,
    FOREIGN KEY (idModele) REFERENCES modeleVoiture(idModele) ON DELETE CASCADE
);

Partie B.2.b : controle.js (verif1)

function verif1() {
    let numPermis = document.getElementById('numPermis').value;
    let nom = document.getElementById('nom').value;
    let prenom = document.getElementById('prenom').value;
    let genreM = document.getElementById('genreM').checked;
    let genreF = document.getElementById('genreF').checked;

    // Contrôle N° Permis: xx/xxxxx
    if (numPermis.length !== 8 || numPermis[2] != '/') {
        alert("Le numéro de permis doit avoir 8 caractères au format xx/xxxxx.");
        return false;
    }
    let part1 = numPermis.substring(0, 2);
    let part2 = numPermis.substring(3, 8);
    if (parseInt(part1) != part1 || parseInt(part2) != part2) {
        alert("Le numéro de permis doit contenir des chiffres.");
        return false;
    }

    // Contrôle Nom: alphabétique, 3 à 20 caractères
    if (nom.length < 3 || nom.length > 20 || alpha(nom) == false) {
        alert("Le nom doit être une chaîne alphabétique de 3 à 20 caractères.");
        return false;
    }

    // Contrôle Prénom: alphabétique, 3 à 20 caractères
    if (prenom.length < 3 || prenom.length > 20 || alpha(prenom) == false) {
        alert("Le prénom doit être une chaîne alphabétique de 3 à 20 caractères.");
        return false;
    }

    // Contrôle Genre: sélection obligatoire
    if (!genreM && !genreF) {
        alert("Veuillez sélectionner un genre.");
        return false;
    }

    return true; // Le formulaire sera soumis
}
function alpha(ch){
    ch = ch.toUpperCase();
    let i = 0;
    while(ch[i] >= 'A' && ch[i] <= 'Z') 
        i++; 
    return i == ch.length; 
}

// Fonction pour changer l'image des notes
function charger() {
    document.getElementById("imgSecurite").src = document.getElementById('securite').value + '.png';
    document.getElementById("imgConduite").src = document.getElementById('conduite').value + '.png';
    document.getElementById("imgConfort").src = document.getElementById('confort').value + '.png';
}

// Fonction de validation du formulaire d'évaluation
function verif2() {
    // Réutilisation du contrôle du N° Permis (voir verif1)
    let numPermis = document.getElementById('numPermis').value;
    if (numPermis.length !== 8 || numPermis[2] != '/' || parseInt(numPermis.substring(0, 2)) != numPermis.substring(0, 2)|| parseInt(numPermis.substring(3, 8))!=numPermis.substring(3, 8)) {
        alert("Format du numéro de permis incorrect.");
        return false;
    }

    // Contrôle Modèle testé
    let idModele = document.getElementById('idModele').selectedIndex;
    if (idModele === 0) {
        alert("Veuillez sélectionner un modèle.");
        return false;
    }

    // Contrôle notes (strictement positif)
    let securite = parseInt(document.getElementById('securite').value);
    let conduite = parseInt(document.getElementById('conduite').value);
    let confort = parseInt(document.getElementById('confort').value);

    if (securite <= 0 || conduite <= 0 || confort <= 0) {
        alert("Les notes de sécurité, conduite et confort doivent être strictement positives.");
        return false;
    }

    // Contrôle case à cocher robot
    let robot = document.getElementById('robot').checked;
    if (!robot) {
        alert("Veuillez cocher la case 'Je ne suis pas un robot'.");
        return false;
    }

    return true;
}

Partie B.3 : evaluation.php

<?php
$con = mysqli_connect("localhost", "root", "", "BDEvaluation");
if (!$con) { die("Erreur de connexion: " . mysqli_error($con)); }

$numPermis = $_POST['numPermis'];
$idModele = $_POST['idModele'];
$securite = $_POST['securite'];
$conduite = $_POST['conduite'];
$confort = $_POST['confort'];

// 1. Vérifier si le testeur existe
$checkTesteur = "SELECT numPermis FROM testeur WHERE numPermis = '$numPermis'";
$resTesteur = mysqli_query($con, $checkTesteur);
if (mysqli_num_rows($resTesteur) == 0) {
    echo "Testeur non inscrit";
    mysqli_close($con);
    exit();
}

// 2. Vérifier si le testeur a déjà évalué ce modèle
$checkEval = "SELECT * FROM evaluation WHERE numPermis = '$numPermis' AND idModele = '$idModele'";
$resEval = mysqli_query($con, $checkEval);
if (mysqli_num_rows($resEval) > 0) {
    echo "Vous avez déjà testé ce modèle";
    mysqli_close($con);
    exit();
}

// 3. Insérer la nouvelle évaluation
$insertQuery = "INSERT INTO evaluation (numPermis, idModele, dateTest, securite, conduite, confort) 
                VALUES ('$numPermis', '$idModele', NOW(), '$securite', '$conduite', '$confort')";

if (mysqli_query($con, $insertQuery)) {
    echo "Evaluation enregistrée avec succès";
} else {
    echo "Erreur: " . mysqli_error($con);
}

mysqli_close($con);
?>

            <?php
            $con = mysqli_connect("localhost", "root", "", "BDEvaluation");
            if (!$con) { die("Erreur de connexion: " . mysqli_error($con)); }

            // La variable $currentYear n'est plus nécessaire

            $sql = "SELECT 
                        m.libelle, 
                        AVG(e.securite) AS moySecurite, 
                        AVG(e.conduite) AS moyConduite, 
                        AVG(e.confort) AS moyConfort, 
                        COUNT(e.idModele) AS nbTests
                    FROM evaluation e, modeleVoiture m
                    WHERE e.idModele = m.idModele AND YEAR(e.dateTest) = Year(Now())
                    GROUP BY e.idModele, m.libelle
                    ORDER BY m.libelle";

            $result = mysqli_query($con, $sql);

            while($row = mysqli_fetch_array($result)) {
                echo "<tr>";
                echo "<td>" . $row['libelle'] . "</td>";
                echo "<td>" . round($row['moySecurite'], 2) . "</td>";
                echo "<td>" . round($row['moyConduite'], 2) . "</td>";
                echo "<td>" . round($row['moyConfort'], 2) . "</td>";
                echo "<td>" . $row['nbTests'] . "</td>";
                echo "</tr>";
            }

            mysqli_close($con);
            ?>
        </tbody>
    </table>
</body>
